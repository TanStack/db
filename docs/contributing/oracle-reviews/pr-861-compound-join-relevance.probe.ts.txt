import { expect, it } from 'vitest'
import { createCollection } from '../../src/collection/index.js'
import { Query, getQueryIR } from '../../src/query/builder/index.js'
import { queriesMatchForCaching } from '../../src/query/compiler/query-equivalence.js'
import { createLiveQueryCollection, and, eq } from '../../src/query/index.js'
import { getQueryIdentity } from '../../src/query/ir-stable-identity.js'
import { mockSyncCollectionOptions } from '../utils.js'

type Row = { id: number; group: number; value: unknown; other: unknown }
let nextSourceId = 0
function source(value: unknown, other: unknown = value) {
  return createCollection(
    mockSyncCollectionOptions<Row>({
      id: `compound-probe-${nextSourceId++}`,
      getKey: (row) => row.id,
      initialData: [{ id: 1, group: 1, value, other }],
    }),
  )
}

// Fixed diagnostic cases: compound conjunction must retain each eq operand's
// established equality. The single-equality control runs on the same sources.
it.each([
  { name: `distinct objects`, left: { n: 1 }, right: { n: 1 }, expected: [] },
  {
    name: `opposite infinities`,
    left: Infinity,
    right: -Infinity,
    expected: [],
  },
  {
    name: `Date versus ISO string`,
    left: new Date(0),
    right: new Date(0).toISOString(),
    expected: [],
  },
  { name: `BigInt equality`, left: 1n, right: 1n, expected: [{ id: 1 }] },
])(
  `compound equality preserves $name`,
  async ({ left: l, right: r, expected }) => {
    const left = source(l)
    const right = source(r)
    const single = createLiveQueryCollection((q) =>
      q
        .from({ left })
        .innerJoin({ right }, ({ left, right }) => eq(left.value, right.value))
        .select(({ left }) => ({ id: left.id })),
    )
    const compound = createLiveQueryCollection((q) =>
      q
        .from({ left })
        .innerJoin({ right }, ({ left, right }) =>
          and(eq(left.group, right.group), eq(left.value, right.value)),
        )
        .select(({ left }) => ({ id: left.id })),
    )
    try {
      await single.preload()
      expect(single.toArray.map(({ id }) => ({ id }))).toEqual(expected)
      await compound.preload()
      expect(compound.toArray.map(({ id }) => ({ id }))).toEqual(expected)
    } finally {
      await Promise.all([
        single.cleanup(),
        compound.cleanup(),
        left.cleanup(),
        right.cleanup(),
      ])
    }
  },
)

it(`additional join conditions distinguish cached queries`, async () => {
  const left = source(1, 2)
  const right = source(1, 3)
  const first = new Query()
    .from({ left })
    .innerJoin({ right }, ({ left, right }) =>
      and(eq(left.group, right.group), eq(left.value, right.value)),
    )
    .select(({ left }) => ({ id: left.id }))
  const second = new Query()
    .from({ left })
    .innerJoin({ right }, ({ left, right }) =>
      and(eq(left.group, right.group), eq(left.other, right.other)),
    )
    .select(({ left }) => ({ id: left.id }))
  const a = createLiveQueryCollection({ query: first })
  const b = createLiveQueryCollection({ query: second })
  try {
    await Promise.all([a.preload(), b.preload()])
    expect(a.toArray.map(({ id }) => ({ id }))).toEqual([{ id: 1 }])
    expect(b.toArray.map(({ id }) => ({ id }))).toEqual([])
    expect
      .soft(getQueryIdentity(getQueryIR(first)))
      .not.toBe(getQueryIdentity(getQueryIR(second)))
    expect
      .soft(queriesMatchForCaching(getQueryIR(first), getQueryIR(second)))
      .toBe(false)
  } finally {
    await Promise.all([
      a.cleanup(),
      b.cleanup(),
      left.cleanup(),
      right.cleanup(),
    ])
  }
})

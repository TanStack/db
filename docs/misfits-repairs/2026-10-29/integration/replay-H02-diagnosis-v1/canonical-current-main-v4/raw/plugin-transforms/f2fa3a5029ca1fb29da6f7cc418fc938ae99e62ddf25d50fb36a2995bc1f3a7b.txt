export function isUnorderable(value) {
  return typeof value === `number` && Number.isNaN(value) || value instanceof Date && Number.isNaN(value.getTime());
}
export const ascComparator = (a, b, opts) => {
  const { nulls } = opts;
  if (a == null && b == null) return 0;
  if (a == null) return nulls === `first` ? -1 : 1;
  if (b == null) return nulls === `first` ? 1 : -1;
  const aUnordered = isUnorderable(a);
  const bUnordered = isUnorderable(b);
  if (aUnordered && bUnordered) return 0;
  if (aUnordered) return 1;
  if (bUnordered) return -1;
  if (typeof a === `string` && typeof b === `string`) {
    if (opts.stringSort === `custom`) {
      return opts.compare(a, b);
    }
    if (opts.stringSort === `locale`) {
      return a.localeCompare(b, opts.locale, opts.localeOptions);
    }
  }
  if (Array.isArray(a) && Array.isArray(b)) {
    for (let i = 0; i < Math.min(a.length, b.length); i++) {
      const result = ascComparator(a[i], b[i], opts);
      if (result !== 0) {
        return result;
      }
    }
    return a.length - b.length;
  }
  if (a instanceof Date && b instanceof Date) {
    return a.getTime() - b.getTime();
  }
  if (isTemporal(a) && isTemporal(b)) {
    return compareTemporalValues(a, b);
  }
  const aIsSymbol = typeof a === `symbol`;
  const bIsSymbol = typeof b === `symbol`;
  if (aIsSymbol && bIsSymbol) {
    if (a === b) return 0;
    return getRuntimeReferenceIdentity(a)[2] - getRuntimeReferenceIdentity(b)[2];
  }
  if (aIsSymbol) return 1;
  if (bIsSymbol) return -1;
  const aIsObject = typeof a === `object`;
  const bIsObject = typeof b === `object`;
  if (aIsObject || bIsObject) {
    if (aIsObject && bIsObject) {
      const aId = getRuntimeReferenceIdentity(a)[2];
      const bId = getRuntimeReferenceIdentity(b)[2];
      return aId - bId;
    }
    if (aIsObject) return 1;
    if (bIsObject) return -1;
  }
  if (a < b) return -1;
  if (a > b) return 1;
  return 0;
};
export const descComparator = (a, b, opts) => {
  return a == null || b == null ? ascComparator(a, b, opts) : ascComparator(b, a, opts);
};
export function indexObjectKind(value) {
  if (isUint8Array(value)) return `Binary`;
  if (isTemporal(value)) {
    const tag = value[Symbol.toStringTag];
    const fields = value;
    return tag === `Temporal.Duration` && (fields.years || fields.months || fields.weeks) ? `Temporal.Duration.calendar` : tag;
  }
  if (Array.isArray(value)) return `Array`;
  return `Object`;
}
export function makeComparator(opts, scope = `values`) {
  const compare = opts.direction === `asc` ? ascComparator : descComparator;
  if (scope !== `index`) return (a, b) => compare(a, b, opts);
  const direction = opts.direction === `asc` ? 1 : -1;
  return (a, b) => {
    if (a != null && b != null) {
      if (a instanceof Date) a = a.getTime();
      if (b instanceof Date) b = b.getTime();
      const aType = typeof a;
      const bType = typeof b;
      if (aType !== bType) return (aType < bType ? -1 : 1) * direction;
      if (aType === `object`) {
        const aKind = indexObjectKind(a);
        const bKind = indexObjectKind(b);
        if (aKind !== bKind) return (aKind < bKind ? -1 : 1) * direction;
        if (aKind === `Binary` || aKind === `Temporal.Duration.calendar` || aKind === `Temporal.PlainMonthDay`) {
          const left = normalizeValue(a);
          const right = normalizeValue(b);
          return (left < right ? -1 : left > right ? 1 : 0) * direction;
        }
      }
    }
    return compare(a, b, opts);
  };
}
export function makeCheckedComparator(compareFn) {
  return (a, b) => {
    const result = compareFn(a, b);
    if (typeof result !== `number` || Number.isNaN(result))
      throw new TypeError(
        devBuild() && process.env.NODE_ENV !== `production` ? `Index comparator must return a number, but returned ${String(result)}` : codedMessage(166, { result })
      );
    return result;
  };
}
export const defaultComparator = makeComparator({
  direction: `asc`,
  nulls: `first`,
  stringSort: `locale`
});
export function isUint8Array(value) {
  return value instanceof Uint8Array || typeof Buffer !== `undefined` && value instanceof Buffer;
}
function areUint8ArraysEqual(a, b) {
  if (a.byteLength !== b.byteLength) {
    return false;
  }
  for (let i = 0; i < a.byteLength; i++) {
    if (a[i] !== b[i]) {
      return false;
    }
  }
  return true;
}

export function normalizeValue(value) {
  if (typeof value === `string`) {
    return value.startsWith(NORMALIZED_KEY_PREFIX) ? normalizedKey(`string`, value) : value;
  }
  if (typeof value !== `object` || value === null) {
    return value;
  }
  if (value instanceof Date) {
    return value.getTime();
  }
  if (isTemporal(value)) {
    const tag = value[Symbol.toStringTag];
    if (tag === "Temporal.Duration") {
      const fields = value;
      return normalizedKey(
        tag,
        durationFields.map((field) => fields[field]).join(",")
      );
    }
    return normalizedKey(tag, value.toString());
  }
  if (isUint8Array(value)) {
    return normalizeBinary(value);
  }
  return value;
}
export function areSameValueZeroEqual(a, b) {
  return a === b || Number.isNaN(a) && Number.isNaN(b);
}
export function compareTemporalValues(a, b) {
  const aTag = a[Symbol.toStringTag];
  const bTag = b[Symbol.toStringTag];
  if (aTag !== bTag) {
    throw new TypeError(
      devBuild() && process.env.NODE_ENV !== `production` ? `Cannot order Temporal values of different types: ${aTag} vs ${bTag}` : codedMessage(167, { aTag, bTag })
    );
  }
  const compare = a.constructor.compare;
  if (typeof compare !== `function`) {
    throw new TypeError(
      devBuild() && process.env.NODE_ENV !== `production` ? `${aTag} has no defined ordering` : codedMessage(168, { aTag })
    );
  }
  return compare(a, b);
}
export function compareValues(a, b) {
  if (isTemporal(a) && isTemporal(b)) {
    return compareTemporalValues(a, b);
  }
  return a < b ? -1 : a > b ? 1 : 0;
}
export function areValuesEqual(a, b) {
  if (a === b) {
    return true;
  }
  if (isUint8Array(a) && isUint8Array(b)) {
    return areUint8ArraysEqual(a, b);
  }
  return false;
}

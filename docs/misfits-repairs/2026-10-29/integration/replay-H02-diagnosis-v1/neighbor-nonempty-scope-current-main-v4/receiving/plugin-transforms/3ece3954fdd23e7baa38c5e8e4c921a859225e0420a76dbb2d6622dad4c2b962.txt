export function toExpression(value) {
  if (value === null || typeof value !== `object` && typeof value !== `function`) {
    return new Value(value);
  }
  const brand = readRefProxyBrand(value);
  if (brand !== void 0) {
    return Array.isArray(brand) ? new PropRef(brand, brand[0], value.__bindingId) : new PropRef(value.__path, value.__sourceAlias, value.__bindingId);
  }
  const name = getWrapperExpressionName(value);
  if (name) {
    throw new Error(
      devBuild() && process.env.NODE_ENV !== `production` ? `${name} cannot be used inside expressions (e.g., coalesce(), eq(), not()). Use ${name} directly as a select field value instead.` : codedMessage(126, { name })
    );
  }
  if (isBasicOrAggregateExpression(value)) {
    return value;
  }
  return new Value(value);
}

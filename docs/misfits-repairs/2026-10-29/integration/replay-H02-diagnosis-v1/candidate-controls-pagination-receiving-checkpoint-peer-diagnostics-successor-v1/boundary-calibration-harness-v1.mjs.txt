function controlledIo(short) {
  let nextFd = 1
  let nextInode = 1
  const files = new Map()
  const fds = new Map()
  function file(path) { const entry=files.get(path);assert(entry,`missing controlled file ${path}`);return entry }
  function fdFile(fd) { const entry=fds.get(fd);assert(entry && !entry.closed);return entry }
  const io = {
    openSync(path,flags) {
      if(flags==='wx') {assert(!files.has(path)); files.set(path,{bytes:Buffer.alloc(0),ino:nextInode++})}
      else assert.equal(flags,'r')
      const entry=file(path);const fd=nextFd++;fds.set(fd,{entry,position:0,closed:false});return fd
    },
    closeSync(fd) {fdFile(fd).closed=true},
    fstatSync(fd) {const {entry}=fdFile(fd);return {isFile:()=>true,dev:1,ino:entry.ino,size:entry.bytes.length}},
    lstatSync(path) {const entry=file(path);return {isFile:()=>true,dev:1,ino:entry.ino,size:entry.bytes.length}},
    writeFileSync(path,bytes,options) {assert.equal(options.flag,'wx');assert(!files.has(path));files.set(path,{bytes:Buffer.from(bytes),ino:nextInode++})},
    writeSync(fd,bytes,offset,length) {const held=fdFile(fd);const count=Math.min(length,short ? 7 : length);held.entry.bytes=Buffer.concat([held.entry.bytes,bytes.subarray(offset,offset+count)]);held.position+=count;return count},
    readSync(fd,bytes,offset,length,position) {const {entry}=fdFile(fd);const count=Math.min(length,entry.bytes.length-position,short ? 7 : length);entry.bytes.copy(bytes,offset,position,position+count);return count},
  }
  return {io,files,fds}
}
function makeTransport(io) {
  return new Function('openSync','fstatSync','writeFileSync','writeSync','closeSync','readSync','lstatSync','TextDecoder',`${writerCode}\n${decoderCode}\n${readerCode}\nreturn {createReceiptWriter, createReceiptReader}`)(io.openSync,io.fstatSync,io.writeFileSync,io.writeSync,io.closeSync,io.readSync,io.lstatSync,TextDecoder)
}
function makeRoute() {
  const state={comparisons:[],cutCalls:0,checkpointCalls:0}
  const build = new Function('state', `
    const publicationEpochs=[{id:'1:held-window',kind:'publication'}]
    let epochIndex=0, activeEpoch, orderedIndex=0, cuts=0
    const epochModels=new Map([['1:held-window',{}]])
    const exposedFacades=new Set(),subscribedFacades=new Set()
    const orderedSteps=[{kind:'baseline-observe',boundary:'held-window'}]
    const model={}
    const deliveryCheck={checkpoint(){state.checkpointCalls++}}
    const checkCompletionBaselineObservation=(actualModel,observation)=>{
      state.cutCalls++
      state.lastObservation={phase:observation.phase,boundary:observation.boundary}
    }
    const compare=(label,invoke)=>{state.comparisons.push(label);invoke()}
    return {route(receipt) {
      if(receipt.phase==='completion-baseline-epoch-start') {${startBranch}}
      else if(receipt.phase==='completion-baseline-epoch-end') {${endBranch}}
      else if(receipt.phase==='completion-baseline-cut') {
        const observation=receipt
        ${cutGuard}
        ${cutCompare}
      }
    },snapshot(){return {activeEpoch,epochIndex,orderedIndex,cuts}}}
  `)
  return {...build(state),state}
}

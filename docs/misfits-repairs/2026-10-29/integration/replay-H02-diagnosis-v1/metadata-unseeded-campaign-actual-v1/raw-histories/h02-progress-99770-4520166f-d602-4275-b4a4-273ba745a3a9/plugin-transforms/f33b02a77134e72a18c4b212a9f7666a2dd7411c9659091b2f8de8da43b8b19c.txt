import { safeRandomUUID } from "./utils/uuid";
import { withCollectionConfigFactory } from "./client.js";
export function localOnlyCollectionOptions(config) {
  const { initialData, onInsert, onUpdate, onDelete, id, ...restConfig } = config;
  const collectionId = id ?? safeRandomUUID();
  const directTypes = /* @__PURE__ */ new Set();
  if (!onInsert) directTypes.add(`insert`);
  if (!onUpdate) directTypes.add(`update`);
  if (!onDelete) directTypes.add(`delete`);
  const syncResult = createLocalOnlySync(initialData, directTypes);
  const wrappedOnInsert = async (params) => {
    if (onInsert) await onInsert(params);
    syncResult.confirmOperationsSync(params.transaction.mutations);
  };
  const wrappedOnUpdate = async (params) => {
    if (onUpdate) await onUpdate(params);
    syncResult.confirmOperationsSync(params.transaction.mutations);
  };
  const wrappedOnDelete = async (params) => {
    if (onDelete) await onDelete(params);
    syncResult.confirmOperationsSync(params.transaction.mutations);
  };
  const acceptMutations = (transaction) => {
    const collectionMutations = transaction.mutations.filter(
      (m) => m.collection.id === collectionId
    );
    if (collectionMutations.length === 0) {
      return;
    }
    syncResult.confirmOperationsSync(
      collectionMutations
    );
  };
  const options = {
    ...restConfig,
    id: collectionId,
    sync: syncResult.sync,
    onInsert: wrappedOnInsert,
    onUpdate: wrappedOnUpdate,
    onDelete: wrappedOnDelete,
    utils: {
      acceptMutations
    },
    startSync: true,
    gcTime: 0
  };
  return withCollectionConfigFactory(
    options,
    () => localOnlyCollectionOptions({
      ...config,
      id: collectionId
    })
  );
}
function createLocalOnlySync(initialData, directTypes) {
  let syncBegin = null;
  let syncWrite = null;
  let syncCommit = null;
  let collection = null;
  const sync = {
    rowUpdateMode: `full`,
    /**
     * Sync function that captures sync parameters and applies initial data
     * @param params - Sync parameters containing begin, write, and commit functions
     * @returns Unsubscribe function (empty since no ongoing sync is needed)
     */
    sync: (params) => {
      const { begin, write, commit, markReady } = params;
      syncBegin = begin;
      syncWrite = write;
      syncCommit = commit;
      collection = params.collection;
      params.collection._state.isLocalOnly = true;
      params.collection._state.localOnlyDirectWrite = {
        types: directTypes,
        write: confirmOperationsSync
      };
      if (params.collection.status !== `ready`) {
        begin();
        initialData?.forEach((item) => {
          write({
            type: `insert`,
            value: item
          });
        });
        commit();
      }
      markReady();
      return () => {
      };
    },
    /**
     * Get sync metadata - returns empty object for local-only collections
     * @returns Empty metadata object
     */
    getSyncMetadata: () => ({})
  };
  const confirmOperationsSync = (mutations) => {
    if (!syncBegin || !syncWrite || !syncCommit) {
      return;
    }
    syncBegin();
    mutations.forEach((mutation) => {
      if (syncWrite) {
        syncWrite({
          type: mutation.type,
          value: mutation.modified
        });
      }
    });
    syncCommit();
  };
  return {
    sync,
    confirmOperationsSync,
    collection
  };
}

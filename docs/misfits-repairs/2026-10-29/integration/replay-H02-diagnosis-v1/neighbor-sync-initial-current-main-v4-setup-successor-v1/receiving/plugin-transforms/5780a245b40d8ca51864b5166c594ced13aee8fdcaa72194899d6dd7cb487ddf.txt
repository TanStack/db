  constructor() {
    super(
      devBuild() && process.env.NODE_ENV !== `production` ? `IndexedDB collection requires a "db" configuration option. Create a database instance using createIndexedDB() and pass it to the collection.` : codedMessage(175)

  }

function isCrossTabMessage(value) {
  if (!value || typeof value !== "object") return false;
  const message = value;
  return (message.type === "data-changed" || message.type === "database-cleared") && typeof message.database === "string" && typeof message.name === "string" && typeof message.tabId === "string" && Array.isArray(message.changedKeys) && message.changedKeys.every(
    (key) => typeof key === "string" || typeof key === "number" && !Number.isNaN(key)
  );
}

export async function createIndexedDB(options) {
  const { name, version, stores, idbFactory, onBlocked } = options;
  if (!stores || stores.length === 0) {
    throw new Error(
      devBuild() && process.env.NODE_ENV !== `production` ? "createIndexedDB requires at least one store in the stores array." : codedMessage(193)
    );
  }
  const storeSet = new Set(stores);
  if (storeSet.size !== stores.length) {
    throw new Error(
      devBuild() && process.env.NODE_ENV !== `production` ? "createIndexedDB stores array contains duplicate store names." : codedMessage(194)
    );
  }
  for (const storeName of stores) {
    if (storeName === VERSIONS_STORE_NAME) {
      throw new Error(
        devBuild() && process.env.NODE_ENV !== `production` ? 'The "_versions" store is reserved for IndexedDB Collection metadata.' : codedMessage(195)
      );
    }
    if (!storeName || typeof storeName !== "string") {
      throw new Error(
        devBuild() && process.env.NODE_ENV !== `production` ? "createIndexedDB stores array contains invalid store names. Each store name must be a non-empty string." : codedMessage(196)
      );
    }
  }
  const db = await openDatabase(
    name,
    version,
    (database) => {
      if (!database.objectStoreNames.contains(VERSIONS_STORE_NAME)) {
        database.createObjectStore(VERSIONS_STORE_NAME);
      }
      for (const storeName of stores) {
        if (!database.objectStoreNames.contains(storeName)) {
          database.createObjectStore(storeName);
        }
      }
    },
    idbFactory,
    onBlocked
  );
  const connection = {
    listeners: /* @__PURE__ */ new Set()
  };
  connections.set(db, connection);
  const close = () => {
    db.close();
    if (connection.error) return;
    connection.error = new Error(
      devBuild() && process.env.NODE_ENV !== `production` ? `IndexedDB connection "${name}" closed. Recreate its Collections with a new database instance.` : codedMessage(197, { name })
    );
    for (const notify of [...connection.listeners]) notify();
  };
  db.addEventListener("versionchange", close);
  db.addEventListener("close", close);
  const frozenStores = Object.freeze([...stores]);
  return Object.freeze({
    db,
    name,
    version: db.version,
    stores: frozenStores,
    idbFactory,
    close
  });
}
function cloneOrReject(operation, clone) {
  try {
    return clone();
  } catch (error) {
    if (error?.name !== "DataCloneError")
      throw error;
    throw new SerializationError(operation, error.message);
  }
}
export function indexedDBCollectionOptions(config) {
  const {
    db: dbInstance,
    name,
    getKey,
    onInsert,
    onUpdate,
    onDelete,
    ...baseCollectionConfig
  } = config;
  if (!dbInstance) {
    throw new DatabaseRequiredError();
  }
  if (!name) {
    throw new NameRequiredError();
  }
  if (name === VERSIONS_STORE_NAME) {
    throw new Error(
      devBuild() && process.env.NODE_ENV !== `production` ? 'The "_versions" store is reserved for IndexedDB Collection metadata.' : codedMessage(198)
    );
  }
  for (const storeName of [name, VERSIONS_STORE_NAME]) {
    if (!dbInstance.db.objectStoreNames.contains(storeName)) {
      throw new ObjectStoreNotFoundError(
        storeName,
        dbInstance.name,
        Array.from(dbInstance.db.objectStoreNames)
      );
    }
  }
  if (!getKey) {
    throw new GetKeyRequiredError();
  }
  const collectionId = baseCollectionConfig.id ?? `indexed-db-collection:${dbInstance.name}:${name}`;
  const connection = connections.get(dbInstance.db);
  const tabId = safeRandomUUID();

  let acceptance = Promise.resolve();

  let broadcastChannel;
  function broadcastChange(changedKeys, type = "data-changed") {
    let channel = broadcastChannel;
    if (!channel) {
      try {
        channel = new BroadcastChannel(`tanstack-db:${dbInstance.name}`);
      } catch {
        return;
      }
    }
    try {
      channel.postMessage({
        type,
        database: dbInstance.name,
        name,
        changedKeys,
        tabId
      });
    } finally {
      if (channel !== broadcastChannel) channel.close();

  function visitVersions(store, visit) {
    return new Promise((resolve, reject) => {
      const request = store.openCursor();
      request.onerror = () => reject(request.error);
      request.onsuccess = () => {
        const cursor = request.result;
        if (!cursor) {
          resolve();
          return;
        }
        try {
          const [storeName] = cursor.key;
          if (storeName < name) {
            cursor.continue([name]);
          } else if (storeName === name) {
            visit(cursor);
            cursor.continue();
          } else {
            resolve();
          }
        } catch (error) {
          reject(error);
        }
      };
    });
  }
  async function persist(mutations, replace = false) {
    const versions = /* @__PURE__ */ new Map();
    await executeTransaction(
      dbInstance.db,
      [name, VERSIONS_STORE_NAME],
      "readwrite",
      async (_, stores) => {
        if (replace) {
          await clear(stores[name]);
          await visitVersions(stores[VERSIONS_STORE_NAME], (cursor) => {
            cursor.delete();
          });
        }
        for (const mutation of mutations) {
          const { key } = mutation;
          if (mutation.type === "delete") {
            stores[name].delete(key);
            stores[VERSIONS_STORE_NAME].delete([name, key]);
          } else {
            const versionKey = safeRandomUUID();
            cloneOrReject(
              mutation.type,
              () => stores[name].put(mutation.modified, key)
            );
            stores[VERSIONS_STORE_NAME].put(
              {
                versionKey
              },
              [name, key]
            );
            versions.set(key, versionKey);
          }
        }
      }
    );
    if (replace) versionCache.clear();
    for (const mutation of mutations) {
      const version = versions.get(mutation.key);
      if (version === void 0) versionCache.delete(mutation.key);
      else versionCache.set(mutation.key, version);
    }
  }
  function confirm(mutations, replace = false) {
    if (!activeSync) return;
    activeSync.begin();
    if (replace) activeSync.truncate({ markReady: !connection?.error });
    for (const mutation of mutations) {
      activeSync.write(
        mutation.type === "delete" ? { type: "delete", key: mutation.key } : { type: "update", value: mutation.modified }
      );
    }
    activeSync.commit();
  }
  async function restore(params, replace) {

      dbInstance.db,
      [name, VERSIONS_STORE_NAME],
      "readonly",
      async (_, stores) => {
        const items = await getAll(stores[name]);
        const versions = /* @__PURE__ */ new Map();

          const [, key] = cursor.key;
          versions.set(key, cursor.value.versionKey);

        return { items, versions };
      }

    for (const [key, version] of snapshot.versions)
      versionCache.set(key, version);

      channel.onmessage = async (event) => {

          return;
        try {
          if (message.type === "database-cleared") {
            await restore(params, true);
            return;
          }
          if (!message.changedKeys.length) return;
          const rows = await executeTransaction(
            dbInstance.db,
            [name, VERSIONS_STORE_NAME],
            "readonly",
            async (_, stores) => {
              const result = [];
              for (const key of message.changedKeys) {
                const [version, value] = await Promise.all([
                  getByKey(stores[VERSIONS_STORE_NAME], [
                    name,
                    key
                  ]),
                  getByKey(stores[name], key)
                ]);
                result.push({ key, version, value });
              }
              return result;
            }
          );
          if (activeSync !== params || connection?.error) return;
          const changes = [];
          for (const { key, version, value } of rows) {
            const cached = versionCache.get(key);
            if (value) {
              if (!version || cached !== version.versionKey) {
                changes.push({
                  type: "update",
                  value
                });
                if (version) versionCache.set(key, version.versionKey);
                else versionCache.delete(key);
              }
            } else {
              changes.push({ type: "delete", key });
              versionCache.delete(key);
            }
          }
          if (changes.length) {
            begin();
            for (const change of changes) write(change);
            commit();
          }
        } catch (error) {
          fail(error);
        }
      };
    } catch {
    }
    void restore(params, false).catch(fail);
    return {
      cleanup: () => {
        connection?.listeners.delete(retire);
        channel?.close();
        if (activeSync === params) {
          activeSync = void 0;
          broadcastChannel = void 0;
          versionCache.clear();
        }
      }
    };
  };
  const acceptMutations = async (transaction) => {
    const mutations = transaction.mutations.filter((m) => m.collection === collection).map((m) => ({
      type: m.type,
      key: getKey(m.modified),
      modified: m.modified
    }));
    if (!mutations.length) return;
    await persist(mutations);
    confirm(mutations);
    broadcastChange(mutations.map((m) => m.key));
  };
  function persistInOrder(transaction, handler) {

  }
  const wrappedOnInsert = (params) => persistInOrder(params.transaction, () => onInsert?.(params));
  const wrappedOnUpdate = (params) => persistInOrder(params.transaction, () => onUpdate?.(params));
  const wrappedOnDelete = (params) => persistInOrder(params.transaction, () => onDelete?.(params));
  const clearObjectStore = async () => {
    await persist([], true);
    confirm([], true);
    broadcastChange([], "database-cleared");
  };
  const getDatabaseInfo = async () => {
    const db = dbInstance.db;
    const info = {
      name: db.name,
      version: db.version,
      objectStores: Array.from(db.objectStoreNames)
    };

      }

  };
  const exportData = () => executeTransaction(
    dbInstance.db,
    name,
    "readonly",
    (_, stores) => getAll(stores[name])
  );

const fixture = '@FIXTURE@'
const pinnedText = (name: string, sha256: string): string => {
  const bytes = readFileSync(`${fixture}/${name}`)
  expect(createHash(`sha256`).update(bytes).digest(`hex`)).toBe(sha256)
  return bytes.toString(`utf8`)
}
const rawHistory = pinnedText(`history.json`, `48b8ce8317a91836b0ae3154d217b442b3491879dc8349e59e9909cad67d6a0b`)
const rawReceipts = pinnedText(`receipts.jsonl`, `6287c60245c2f99f9045d21fbb3d48eb5e2b417137db3787f45bacf7bc2cc13a`)
const rawSupervisor = pinnedText(`supervisor.json`, `fac0489b928bec0c0c2d84b17e202b1ddc5b0db352f6f4cf0478a8e7fa3bd824`)
const history = JSON.parse(rawHistory) as MetadataHistory
const original = rawReceipts.trimEnd().split(`\n`).map((line) => JSON.parse(line) as ProgressReceipt)
const outcome = (JSON.parse(rawSupervisor) as { outcome: ProgressOutcome }).outcome
type Stage = `root-payload` | `downstream-payload` | `metadata` | `outcome` | `finish` | `reach`
type Failure = { sequence: unknown; phase: string; stage: Stage; error: unknown }
type Assertion = Error & { actual?: unknown; expected?: unknown }

// This loop copies only the frozen registration's checker composition. Every
// expectation, model transition and finalizer comes from its unchanged owner.
// The first thrown check stops the replay; no later comparison receives credit.
function replay(receipts: Array<ProgressReceipt>) {
  const reach = createOracleReach(`metadata.root-omission.checker-calibration`, metadataHistoryStates(history), { replaying: false })
  reach.beginHistory()
  const metadata = createMetadataReceiptCheck(history, (state) => reach.reach(state))
  const root = createProgressReceiptCheck(history)
  const downstream = createProgressReceiptCheck(history)
  const completed: Array<{ sequence: unknown; phase: string; stage: Stage }> = []
  let current = { sequence: undefined as unknown, phase: `before-receipts`, stage: `root-payload` as Stage }
  const invoke = (receipt: ProgressReceipt, stage: Stage, check: () => void) => {
    current = { sequence: receipt.sequence, phase: receipt.phase, stage }
    check()
    completed.push({ ...current })
  }
  try {
    for (const receipt of receipts) {
      invoke(receipt, `root-payload`, () => root(receipt))
      if (receipt.phase === `downstream-change`)
        invoke(receipt, `downstream-payload`, () => downstream({ ...receipt, phase: `public-change` }))
      else if ([`reload-start`, `cleanup-start`, `followup-start`].includes(receipt.phase))
        invoke(receipt, `downstream-payload`, () => downstream(receipt))
      else if ([`final-publication`, `followup-publication`].includes(receipt.phase))
        invoke(receipt, `downstream-payload`, () => downstream({ ...receipt, rows: receipt.downstream as typeof receipt.rows }))
      invoke(receipt, `metadata`, () => metadata(receipt))
    }
    const end = { phase: `after-receipts`, sequence: 206 }
    invoke(end, `outcome`, () => checkQueuedAnswerProgress(history, outcome))
    invoke(end, `finish`, () => metadata.finish())
    invoke(end, `reach`, () => reach.assertReached())
    return { completed, counts: Object.fromEntries(reach.counts()), failure: undefined }
  } catch (error) {
    return { completed, counts: Object.fromEntries(reach.counts()), failure: { ...current, error } as Failure }
  }
}

it(`rejects an omitted root previous value at the shared metadata delta comparison`, () => {
  const evidence: Record<string, unknown> = {
    kind: `checker calibration only`, pid: process.pid, parentPid: process.ppid,
    history, originalReceiptCount: original.length,
    law: `complete previous public row`, productionMutant: false,
  }
  let primaryFailure: unknown
  try {
    const healthy = replay(original)
    evidence.healthy = healthy
    if (healthy.failure) throw healthy.failure.error
    expect(Object.values(healthy.counts)).toEqual(metadataHistoryStates(history).map(() => 1))
    const changed = structuredClone(original)
    const target = changed.filter((receipt) => receipt.sequence === 182)
    expect(target).toHaveLength(1)
    expect(target[0]!.phase).toBe(`public-change`)
    const changes = target[0]!.changes as Array<Record<string, unknown>>
    expect(changes).toHaveLength(1)
    const change = changes[0]!
    const previousValue = change.previousValue as Record<string, unknown>
    const value = change.value as Record<string, unknown>
    const baseline = original.find((receipt) => receipt.phase === `final-publication`)!
    expect(baseline.sequence).toBe(161)
    expect(change.type).toBe(`update`)
    expect(change.key).toBe(`metadata-root-3`)
    expect(change.key).toBe(previousValue.$key)
    expect(change.key).toBe(value.$key)
    expect(previousValue.id).toBe(3)
    expect(value.id).toBe(3)
    expect(previousValue.rank).toBe(30)
    expect(value.rank).toBe(35)
    expect([previousValue]).toEqual(baseline.rows)
    expect(previousValue.$collectionId).toBe(value.$collectionId)
    delete change.previousValue
    // The mutation footprint is one absent property in a cloned observation.
    // Restoring it reconstructs every original receipt, field and occurrence.
    const restored = structuredClone(changed)
    const restoredTarget = restored.find((receipt) => receipt.sequence === 182)!
    const restoredChanges = restoredTarget.changes as Array<Record<string, unknown>>
    restoredChanges[0]!.previousValue = previousValue
    expect(restored).toEqual(original)
    evidence.targetBefore = original.find((receipt) => receipt.sequence === 182)
    evidence.targetAfter = target[0]
    const hostile = replay(changed)
    const failure = hostile.failure
    const error = failure?.error as Assertion | undefined
    evidence.hostile = { ...hostile, failure: failure ? { ...failure, error: error ? {
      name: error.name, message: error.message, stack: error.stack,
      actual: error.actual, expected: error.expected,
    } : failure.error } : undefined }
    expect(failure).toBeDefined()
    expect(failure?.sequence).toBe(185)
    expect(failure?.phase).toBe(`followup-publication`)
    expect(failure?.stage).toBe(`metadata`)
    expect(error?.name).toBe(`AssertionError`)
    expect(error?.stack).toContain(`checkMetadataPublications`)
    expect(error?.stack).toContain(`virtual-row-fields-publication-oracle.ts`)
    expect(error?.actual).toEqual(changes)
    expect(error?.expected).toEqual((original.find((receipt) => receipt.sequence === 182)!).changes)
    expect(hostile.completed).toContainEqual({ sequence: 182, phase: `public-change`, stage: `root-payload` })
    expect(hostile.completed).toContainEqual({ sequence: 182, phase: `public-change`, stage: `metadata` })
    expect(hostile.completed).toContainEqual({ sequence: 183, phase: `downstream-change`, stage: `downstream-payload` })
    expect(hostile.completed).toContainEqual({ sequence: 184, phase: `followup-settled`, stage: `metadata` })
    expect(hostile.completed).toContainEqual({ sequence: 185, phase: `followup-publication`, stage: `downstream-payload` })
    expect(hostile.counts[`reload-baseline-compared`]).toBe(1)
    expect(hostile.counts[`followup-update-root`]).toBe(0)
    expect(hostile.completed.some((entry) => entry.stage === `finish`)).toBe(false)
    evidence.classification = `intended shared root metadata delta assertion rejection`
  } catch (error) {
    primaryFailure = error
    evidence.primaryFailure = error instanceof Error ? { name: error.name, message: error.message, stack: error.stack } : String(error)
    throw error
  } finally {
    try {
      writeFileSync(`calibration-result.json`, JSON.stringify(evidence, null, 2) + `\n`, { flag: `wx`, mode: 0o600 })
    } catch (captureFailure) {
      throw new AggregateError([captureFailure], `Checker calibration evidence capture failed`, { cause: primaryFailure })
    }
  }
}, 10_000)
